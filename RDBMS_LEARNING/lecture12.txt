(
emp_id int not null,
month int not null,
year int not null,
total_present int not null,
total_absent int not null,
total_leave int not null,
primary key (emp_id,month,year),
foreign key (emp_id) references employee(emp_id)
);

create trigger employee_attendance_insert after insert on employee_attendance for each row
Begin
if new.status="P" or new.status="p"
then
update employee_attendance_summary set total_present=total_present+1 where emp_id=new.emp_id and month(new.date_of_attendance)=month and year(new.date_of_attendance)=year;
end if;
if new.status="A" or new.status="a"
then
update employee_attendance_summary set total_absent=total_absent+1 where emp_id=new.emp_id and month(new.date_of_attendance)=month and year(new.date_of_attendance)=year;
end if;
if new.status="L" or new.status="l"
then
update employee_attendance_summary set total_leave=total_leave+1 where emp_id=new.emp_id and month(new.date_of_attendance)=month and year(new.date_of_attendance)=year;
end if;
End; //

Delimiter;
