//Self Join
create table employee_boss
(
emp_id int not null,
boss_emp_id int not null,
primary key (emp_id,boss_emp_id),
foreign key (emp_id) references employee(emp_id),
foreign key (boss_emp_id) references employee(emp_id)
);

e2.first_name,e2.last_name from employee e1

on e1.emp_id=employee_boss.emp_id

on e2.emp_id=employee_boss.boss_emp_id

create procedure apply_increments(in increment int,in hra_factor int,in basic_salary_factor int)
begin
set autocommit=0;
declare total_amount int;
start transaction;
update employee_salary set hra=hra+increment;
select sum(hra) into total_amount from employee_salary
if total_amount < hra_factor then
update employee_salary set basic_salary=basic_salary+increment;
end if;
select sum(basic_salary+hra) into total_amount from employee_salary;
if total_amount > basic_salary_factor then
rollback;
else
commit;
end if;
set autocommit=1;
end;
delimiter ;
