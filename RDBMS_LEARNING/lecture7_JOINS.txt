4. Get emp_id and name of all American employees
select employee.emp_id,employee.first_name,employee.last_name from employee inner join
employee_address on employee.emp_id=employee_address.emp_id inner join
city on employee_address.city_code=city.code inner join
state on city.state_code=state.code inner join
country on state.country_code=country.code
where country.name="USA"

5. Get emp_id and name of all Indian MALE employees
select employee.emp_id,employee.first_name,employee.last_name,employee.gender
 from employee inner join
employee_address on employee.emp_id=employee_address.emp_id inner join
city on employee_address.city_code=city.code inner join
state on city.state_code=state.code inner join
country on state.country_code=country.code
where country.name="India" and employee.gender='M'

6. Get emp_id and name of all American FEMALE employees
select employee.emp_id,employee.first_name,employee.last_name,employee.gender
 from employee inner join

city on employee_address.city_code=city.code inner join
state on city.state_code=state.code inner join
country on state.country_code=country.code
where country.name="USA" and employee.gender='F'

8. Get emp_id and name of all employee whose addr is of only 2 lines
select employee.emp_id,employee.first_name,employee.last_name from employee
inner join employee_address on employee.emp_id=employee_address.emp_id
where (employee_address.line1 is not null and employee_address.line2 is not null
and employee_address.line3 is null)
